tests/test_attendance.py

# tests/test_attendance.py
import pytest
from datetime import datetime, date
from app.services.attendance import AttendanceService
from app.models import Employee, AttendanceRecord
from app.exceptions import AlreadyClockedInError, NotClockedInError

class TestClockIn:
    def test_clock_in_success(self, db_session, sample_employee):
        """정상 출근: AttendanceRecord 생성 + clock_in_at 기록"""
        service = AttendanceService(db_session)
        record = service.clock_in(
            employee_id=sample_employee.id,
            timestamp=datetime(2026, 5, 26, 9, 0, 0)
        )
        assert record.employee_id == sample_employee.id
        assert record.clock_in_at.hour == 9
        assert record.clock_out_at is None
        assert record.status == "normal"

    def test_clock_in_duplicate(self, db_session, sample_employee, existing_clock_in):
        """당일 중복 출근: AlreadyClockedInError(409) 발생"""
        service = AttendanceService(db_session)
        with pytest.raises(AlreadyClockedInError):
            service.clock_in(
                employee_id=sample_employee.id,
                timestamp=datetime(2026, 5, 26, 10, 0, 0)
            )


class TestClockOut:
    def test_clock_out_success(self, db_session, sample_employee, existing_clock_in):
        """정상 퇴근: work_minutes = 540분(9시간) 계산"""
        service = AttendanceService(db_session)
        record = service.clock_out(
            employee_id=sample_employee.id,
            timestamp=datetime(2026, 5, 26, 18, 0, 0)
        )
        assert record.clock_out_at is not None
        assert record.work_minutes == 540

    def test_clock_out_without_clock_in(self, db_session, sample_employee):
        """출근 없이 퇴근: NotClockedInError(409) 발생"""
        service = AttendanceService(db_session)
        with pytest.raises(NotClockedInError):
            service.clock_out(
                employee_id=sample_employee.id,
                timestamp=datetime(2026, 5, 26, 18, 0, 0)
            )


class TestLeaveRequest:
    def test_submit_leave_annual(self, db_session, sample_employee):
        """연차 신청: LeaveRequest 생성, status=pending"""
        from app.services.leave import LeaveService
        from datetime import datetime
        service = LeaveService(db_session)
        request = service.submit(
            employee_id=sample_employee.id,
            leave_type="annual",
            start_at=datetime(2026, 6, 2, 9, 0, 0),
            end_at=datetime(2026, 6, 2, 18, 0, 0),
            reason="개인 사유"
        )
        assert request.status == "pending"
        assert request.leave_type == "annual"

    def test_submit_leave_invalid_date(self, db_session, sample_employee):
        """종료일 < 시작일: InvalidDateRangeError(422) 발생"""
        from app.services.leave import LeaveService
        from app.exceptions import InvalidDateRangeError
        service = LeaveService(db_session)
        with pytest.raises(InvalidDateRangeError):
            service.submit(
                employee_id=sample_employee.id,
                leave_type="annual",
                start_at=datetime(2026, 6, 5, 9, 0, 0),
                end_at=datetime(2026, 6, 2, 18, 0, 0),  # 종료 < 시작
                reason="오류 케이스"
            )

    def test_approve_leave_by_manager(self, db_session, sample_employee, sample_manager, pending_leave):
        """팀장 승인: status=approved, processed_at 기록"""
        from app.services.leave import LeaveService
        service = LeaveService(db_session)
        result = service.approve(
            request_id=pending_leave.id,
            approver_id=sample_manager.id,
            approved=True,
            comment="승인합니다."
        )
        assert result.status == "approved"
        assert result.approved_by == sample_manager.id
        assert result.processed_at is not None

    def test_approve_leave_by_employee_forbidden(self, db_session, sample_employee, pending_leave):
        """직원이 승인 시도: ForbiddenError(403) 발생"""
        from app.services.leave import LeaveService
        from app.exceptions import ForbiddenError
        service = LeaveService(db_session)
        with pytest.raises(ForbiddenError):
            service.approve(
                request_id=pending_leave.id,
                approver_id=sample_employee.id,  # 직원 — 권한 없음
                approved=True,
                comment=""
            )

# tests/conftest.py
import pytest
from datetime import datetime
from sqlalchemy import create_engine
from sqlalchemy.orm import Session
from app.models.base import Base
from app.models import Employee, AttendanceRecord, LeaveRequest
from app.enums import Role

@pytest.fixture(scope="function")
def db_session():
    """인메모리 SQLite 세션 — 각 테스트마다 독립 DB"""
    engine = create_engine(
        "sqlite:///:memory:",
        connect_args={"check_same_thread": False}
    )
    Base.metadata.create_all(engine)
    with Session(engine) as session:
        yield session
    Base.metadata.drop_all(engine)

@pytest.fixture
def sample_employee(db_session):
    """기본 직원 픽스처 (ROLE-01)"""
